\begin{problem}
    (Adapted from Kiran S. Kedlaya, \emph{Geometry Unbound}, Problem 4.3.2.)
    A circle is tangent to the sides $AB$, $BC$, $CD$, $DA$ of quadrilateral
    $ABCD$ at $M$, $N$, $P$, $Q$, respectively. Prove that $AC$, $BD$, $MP$,
    and $NQ$ pass through a common point.
\end{problem}

\begin{problem}
    Let $n\geq 4$. Let $S=[n]$ denote the set $\{1,2,3,...,n\}$ denote a set of ``points'', do the following families of subsets of $S$ (i.e. ``lines'') satisfy Hilbert's incidence axioms for geometry. 
    \begin{enumerate}
        \item $\binom{[n]}{2}=\{X\subseteq [n]: \vert X\vert =2\}$ (i.e. the set of all two element subsets of $[n]$.)
        \item $\binom{[n]}{n-1}=\{X\subseteq [n]: \vert X\vert =n-1\}$ (i.e. the set of all $n-1$ element subsets of $[n]$.)
        \item Here the lines consist of sets of the form $\{1,k\}$ where $2\leq k\leq n$ and the set $\{2,3,4,...,n\}$.
    \end{enumerate}
\end{problem}
