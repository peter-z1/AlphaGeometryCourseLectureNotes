\section{Mathematical Problems}

\begin{problem}
    Consider two circles $O_1$ and $O_2$ which do not intersect. Let $l_1$ and $l_2$ denote the two mutual tangents of these circles, and say $l_1$ intersects circle $O_1$ at $A$ and $O_2$ at $B$, and $l_2$ intersects circle $O_1$ at $C$ and $O_2$ at $D$. Let $M$ be the midpoint of $A$ and $B$, and $N$ the midpoint of $C$ and $D$. Show that the line through $M$ and $N$ is the radial axis of the two circles $O_1$ and $O_2$.
\end{problem}

\begin{problem}
    Let $O_1$ and $O_2$ be circles such that $O_2$ is entirely inside of $O_1$, construct (i.e., only using a straight edge and a compass) the radial axis of these two circles. 
\end{problem}

\begin{problem}
   Assume that $ABCD$ is a quadrangle inscribed in a circle. Prove that
\[
AB\cdot CD + BC \cdot AD = AC \cdot BD.
\]
\end{problem}

\begin{problem}

\end{problem}
