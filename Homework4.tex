\section{Mathematical Problems}

\begin{problem}
Let $D,E,F$ be points on the segments $BC$, $CA$, $AB$ respectively. Show that the circumcircles of $\Delta AEF$, $\Delta BDF$, $\Delta CDE$ intersect at a single point $M$.
\end{problem}

\begin{problem}
Let $M$ be the midpoint of $BC$, and let $G$ be the centroid of the triangle $ABC$. Show that $AG/MG=2$.  
\end{problem}

\begin{problem}
Let $O$ and $G$ be the circumcenter and centroid of the triangle $ABC$ respectively. Let $M$ be the midpoint of $BC$. Let $P$ be a point on the ray $\vec{OG}$ such hat $PG/GO =2$. Show that $AP$ is parallel to $OM$. 
\end{problem}

\begin{problem}
    Let $H$ be the orthocenter of the triangle $ABC$. Show that $O,H,G$ are colinear.
\end{problem}
