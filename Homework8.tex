\section{Mathematical Problems}

\begin{problem}
    Recall the proposition that we were proving from class on Wednesday.

    \emph{Source note.} Adapted from Hartshorne, \emph{Geometry: Euclid and
    Beyond}, Proposition 7.1.

    \textbf{Proposition (Plane Separation).} For any line $l$, the points not
    lying on $l$ can be partitioned into two nonempty sets $S_1$ and $S_2$
    satisfying:
    \begin{enumerate}[label=(\alph*)]
        \item Two points $A$, $B$ not on $l$ belong to the same set iff the segment $\overline{AB}$ does not intersect $l$.
        \item Two points $A$, $B$ not on $l$ belong to opposite sets iff the segment $\overline{AB}$ intersects $l$. 
    \end{enumerate}

    We defined a relation $\sim$ by $A\sim B$ if
    $\overline{AB}\cap l=\emptyset$, and $A\not\sim B$ if
    $\overline{AB}\cap l\neq\emptyset$.

    \begin{enumerate}
        \item Assume that $A,B,C$ are collinear in this case prove that if $A\sim B$ and $B\sim C$, then $A\sim C$. This will complete the proof that $\sim$ is an equivalence relation. 

        \item Show that there are only $2$ equivalence classes determined by $\sim$. That is show that if $A\not\sim B$ and $B\not\sim C$, then $A\sim C$. 
    \end{enumerate}
\end{problem}
